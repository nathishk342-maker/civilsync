STEP 2 — script.js

JavaScript
const materials = {
    "Cement": {
        rate: 420,
        unit: "bag"
    },

    "M-Sand": {
        rate: 1800,
        unit: "m³"
    },

    "20mm Aggregate": {
        rate: 1600,
        unit: "m³"
    },

    "Bricks": {
        rate: 8500,
        unit: "1000 bricks"
    },

    "Steel": {
        rate: 65,
        unit: "kg"
    }
};


function updateMaterialRate() {

    let material =
        document.getElementById("materialName").value;

    if (material === "") {
        document.getElementById("materialRate").value = "";
        document.getElementById("materialUnit").innerHTML = "Unit: -";
        return;
    }

    document.getElementById("materialRate").value =
        materials[material].rate;

    document.getElementById("materialUnit").innerHTML =
        "Unit: " + materials[material].unit;
}


function calculateCost() {

    let material =
        document.getElementById("materialName").value;

    let quantity =
        parseFloat(
            document.getElementById("materialQuantity").value
        );

    if (material === "") {
        document.getElementById("costResult").innerHTML =
            "Please select a material.";
        return;
    }

    if (isNaN(quantity) || quantity <= 0) {
        document.getElementById("costResult").innerHTML =
            "Please enter a valid quantity.";
        return;
    }

    let rate = materials[material].rate;

    let total = quantity * rate;

    document.getElementById("costResult").innerHTML =
        "<strong>" + material + "</strong><br>" +
        "Quantity: " + quantity + " " + materials[material].unit +
        "<br>Rate: ₹" + rate.toLocaleString("en-IN") +
        "<br><br><strong>Estimated Cost: ₹" +
        total.toLocaleString("en-IN") +
        "</strong>";
}
